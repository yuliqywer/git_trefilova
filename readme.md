git init - инициализация репозитория
git add - добавить файлы
git commit - фиксация изменений
git status - показывает текущее состояние репозитория
git branch - управление ветками в репозитории
git switch - переключение между ветками (новые версии)
git push - отправляет локальные изменения из репозитория в удаленный
git merge - слить указанную ветку в текущую
git branch -d - удаляет локальную ветку
git checkout - переключение между ветками (старые версии)
git log - просмотреть историю коммитов
